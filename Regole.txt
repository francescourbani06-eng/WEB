        import matplotlib.pyplot as plt

        # Nuovo grafico
        plt.figure()

        # Grafici principali
        plt.plot(x, y)              # grafico a linee
        plt.bar(x, y)               # grafico a barre
        plt.hist(x)                 # istogramma
        plt.hist(x, bins=10)        # istogramma con 10 intervalli
        plt.scatter(x, y)           # grafico a punti

        # Titolo e assi
        plt.title("Titolo")
        plt.xlabel("Asse x")
        plt.ylabel("Asse y")

        # Personalizzazione semplice
        plt.grid()                  # mostra griglia
        plt.xlim(a, b)              # limiti asse x
        plt.ylim(a, b)              # limiti asse y

        # Più linee nello stesso grafico
        plt.plot(x, y1, label="A")
        plt.plot(x, y2, label="B")
        plt.legend()                # mostra legenda

        # Visualizzare / salvare
        plt.show()
        plt.savefig("grafico.png")
